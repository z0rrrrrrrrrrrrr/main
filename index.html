<body>

    <!-- CHANGE 'character.png' TO THE EXACT NAME OF YOUR IMAGE FILE -->

    <img id="bouncing-image" src=“022_”11.jpg alt="Bouncing Character">


    <script>

        const img = document.getElementById('bouncing-image');

        

        let x = 0;

        let y = 0;

        let vx = 2; // Horizontal speed

        let vy = 2; // Vertical speed

        let scaleX = 1; // 1 for facing right, -1 for facing left

        

        // Wait for the image to load before starting the animation

        img.onload = function() {

            animate();

        };

        

        // If the image is already cached/loaded

        if (img.complete) {

            animate();

        }

        

        function animate() {

            const maxX = window.innerWidth - img.offsetWidth;

            const maxY = window.innerHeight - img.offsetHeight;

            

            x += vx;

            y += vy;

            

            // Hit right wall -> flip to face left

            if (x >= maxX) {

                vx = -vx;

                scaleX = -1;

                x = maxX; 

            }

            

            // Hit left wall -> flip to face right

            if (x <= 0) {

                vx = -vx;

                scaleX = 1;

                x = 0; 

            }

            

            // Hit bottom or top -> just bounce, keep current horizontal flip

            if (y >= maxY || y <= 0) {

                vy = -vy;

                y = Math.max(0, Math.min(y, maxY));

            }

            

            // Apply position and horizontal flip

            img.style.transform = `translate(${x}px, ${y}px) scaleX(${scaleX})`;

            

            requestAnimationFrame(animate);

        }

        

        // Handle window resize so it doesn't get stuck off-screen

        window.addEventListener('resize', function() {

            const maxX = window.innerWidth - img.offsetWidth;

            const maxY = window.innerHeight - img.offsetHeight;

            x = Math.min(x, maxX);

            y = Math.min(y, maxY);

        });

    </script>
